def _focus_minute_fullscreen_window(self, clicked_dt):
        """V12 新增：把"分钟线全屏窗口"激活/打开，并跳到 clicked_dt 的中心。

        关键点：
        - 复用 Monitor 已有的 ``self._fullscreen_windows`` 全屏窗口列表；
        - 在列表里找 interval 是"分钟"的那一个（minute/5m/1m/15m/30m/60m），
          如果找不到，自动帮用户开一个新的；
        - 调 ``raise_()`` + ``activateWindow()`` 让它置顶（不会被日线全屏挡住）；
        - 调它的 chart.focus_datetime(clicked_dt) 把对应日的分钟K线居中显示。
        """
        try:
            from vnpy.trader.constant import Interval

            # 1) 收集"全屏窗口列表"
            fullscreen_windows = getattr(self, '_fullscreen_windows', None) or []
            # 兼容老路径：直接在 attribute 上找
            if not fullscreen_windows:
                fullscreen_windows = []

            # 2) 在已开的全屏窗口中找"分钟线"那个
            minute_fs = None
            for w in fullscreen_windows:
                # _KlineFullscreenWindow 暴露了 _interval 属性
                iv = getattr(w, '_interval', None)
                if iv is None:
                    continue
                if iv in (Interval.MINUTE, Interval.MINUTE_5, Interval.MINUTE_15,
                         Interval.MINUTE_30, Interval.HOUR):
                    minute_fs = w
                    break

            # 3) 如果没找到，自动打开一个"分钟线全屏窗口"
            if minute_fs is None:
                print(f"[联动V12] 未找到分钟线全屏窗口，尝试自动打开")
                try:
                    # 复用在 widget.py 里定义的 KlineViewTab.open_fullscreen 接口
                    # 找当前 Monitor 的 _minute_panel._kline_tab（_KlineViewTab 实例）
                    minute_panel = getattr(self, '_minute_panel', None)
                    kline_tab = getattr(minute_panel, '_kline_tab', None) if minute_panel else None
                    if kline_tab is None:
                        print(f"[联动V12] 找不到 _minute_panel._kline_tab，无法自动开分钟全屏")
                        return
                    # 多数实现里 KlineViewTab 提供 open_fullscreen() 方法
                    if not hasattr(kline_tab, 'open_fullscreen'):
                        print(f"[联动V12] _kline_tab 没有 open_fullscreen 接口")
                        return
                    kline_tab.open_fullscreen()
                    # open_fullscreen 内部会把新窗口 append 到 self._fullscreen_windows
                    fullscreen_windows = getattr(self, '_fullscreen_windows', None) or []
                    for w in fullscreen_windows:
                        iv = getattr(w, '_interval', None)
                        if iv in (Interval.MINUTE, Interval.MINUTE_5, Interval.MINUTE_15,
                                 Interval.MINUTE_30, Interval.HOUR):
                            minute_fs = w
                            break
                except Exception as _auto_exc:
                    print(f"[联动V12] 自动打开分钟全屏窗口失败: {_auto_exc}")
                    import traceback
                    traceback.print_exc()
                    return

            if minute_fs is None:
                print(f"[联动V12] 仍然找不到分钟线全屏窗口，放弃")
                return

            # 4) 把分钟线全屏窗口置顶（不被日线全屏挡住）
            try:
                if minute_fs.isMinimized():
                    minute_fs.showNormal()
                minute_fs.showMaximized()
                minute_fs.raise_()
                minute_fs.activateWindow()
            except Exception as _raise_exc:
                print(f"[联动V12] raise_/activateWindow 失败: {_raise_exc}")

            # 5) 跳到 clicked_dt 的中心
            try:
                chart = getattr(minute_fs, '_chart', None)
                if chart is not None and hasattr(chart, 'focus_datetime'):
                    # focus_datetime(clicked_dt) 由 KlineChartWidget 提供：
                    # 会把 x 范围设置为以 clicked_dt 为中心的窗口（默认 60 根分钟线）
                    chart.focus_datetime(clicked_dt, completed_daily=False)
                    print(f"[联动V12] 跳转到分钟线全屏 {clicked_dt.date()} 中心完成")
                else:
                    print(f"[联动V12] 分钟线全屏没有 chart.focus_datetime 接口")
            except Exception as _focus_exc:
                print(f"[联动V12] focus_datetime 失败: {_focus_exc}")
                import traceback
                traceback.print_exc()

        except Exception as e:
            print(f"[联动V12] _focus_minute_fullscreen_window 失败: {e}")
            import traceback
            traceback.print_exc()
